### Fonction `setLanguage()` dans `index.html`

```javascript
// Inclure translations.js AVANT ce script
<script src="config/translations.js"></script>

<script>
    // Langue courante
    let currentLang = 'fr';
    
    // Obtenir langue préférée
    function getCurrentLanguage() {
        // 1. Vérifier localStorage
        const saved = localStorage.getItem('lang');
        if (saved) return saved;
        
        // 2. Détecter langue navigateur
        const browserLang = navigator.language.toLowerCase();
        if (browserLang.startsWith('es')) return 'es';
        
        // 3. Défaut : français
        return 'fr';
    }
    
    // Fonction de traduction (pour JS)
    function t(key) {
        const keys = key.split('.');
        let value = translations[currentLang];
        
        for (const k of keys) {
            value = value[k];
            if (!value) return key; // Fallback si clé inexistante
        }
        
        return value;
    }
    
    // Changer langue
    function setLanguage(lang) {
        currentLang = lang;
        localStorage.setItem('lang', lang);
        
        // Traduire tous les éléments avec data-i18n
        document.querySelectorAll('[data-i18n]').forEach(el => {
            const key = el.getAttribute('data-i18n');
            el.textContent = t(key);
        });
        
        // Traduire tous les placeholders avec data-i18n-placeholder
        document.querySelectorAll('[data-i18n-placeholder]').forEach(el => {
            const key = el.getAttribute('data-i18n-placeholder');
            el.placeholder = t(key);
        });
    }
    
    // Switch langue avec boutons
    function switchLanguage(lang) {
        setLanguage(lang);
        
        // Mettre à jour boutons
        document.querySelectorAll('.lang-btn').forEach(btn => {
            btn.classList.remove('active');
        });
        document.getElementById(`lang-${lang}`).classList.add('active');
    }
    
    // Initialiser au chargement
    document.addEventListener('DOMContentLoaded', () => {
        const lang = getCurrentLanguage();
        switchLanguage(lang);
    });
</script>
```

## 🎓 POINTS D'ATTENTION

### ⚠️ Pièges à éviter
1. **Oublier `data-i18n`** sur un élément → restera en français
2. **Clé inexistante** dans `translations.js` → affichera la clé elle-même
3. **Textes dynamiques** (générés en JS) → utiliser fonction `t()`
4. **Accents espagnols** → bien utiliser á, é, í, ó, ú, ñ
5. **Longueur textes ES** → parfois + long que FR, vérifier mise en page

### ✅ Bonnes pratiques
1. **Tester switch FR/ES régulièrement** pendant développement
2. **Commencer par traduire** les textes les + visibles (titre, boutons)
3. **Utiliser des clés descriptives** : `criteria.security` plutôt que `c1`
4. **Sauvegarder souvent** (commits intermédiaires)
5. **Tester sur mobile** (responsive)
